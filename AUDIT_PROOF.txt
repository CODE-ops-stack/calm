class StrictPlayStoreAppVerifier(context: Context) {
    private val deContext = context.createDeviceProtectedStorageContext()
    private val pm = deContext.packageManager
    private val dpm = deContext.getSystemService(DevicePolicyManager::class.java)
    private val admin = ComponentName(deContext, SwatantraAdminReceiver::class.java)

            ApplicationInfo.CATEGORY_VIDEO,
            ApplicationInfo.CATEGORY_AUDIO,
            ApplicationInfo.CATEGORY_IMAGE,

            "academy", "classes", "lecture", "notes", "pdf", "dictionary", "flashcard", "anki", "learn", "coding", "python", "c++"

        /**
         * Pure static 5-gate anti-cheat educational verification logic.
         * Evaluates package, label, category, source installer, and browser/VPN intent probe results.
         */
        fun isStaticMetadataEducational(
            pkg: String,
            appLabel: String,
            category: Int,
            installer: String?,
            hasBrowserActivity: Boolean,
            hasVpnService: Boolean
        ): Boolean {
            // Gate A: Curated Study Registry (immediate allow)
            if (pkg in curatedStudyPackages) {
                return true
            }

            // Gate B: Official Play Store Source Check
            if (installer != "com.android.vending") {
                return false
            }

            // Gate C: Anti-Browser & Anti-VPN Intent Probe
            if (hasBrowserActivity || hasVpnService) {
                return false
            }

            // Gate D: OS Category & Entertainment Keyword Blacklist
            if (category in blacklistedCategories) {
                return false
            }

            val lowerPkg = pkg.lowercase()
            val lowerLabel = appLabel.lowercase()

            if (entertainmentTokens.any { lowerPkg.contains(it) || lowerLabel.contains(it) }) {
                return false
            }

            // Gate E: Educational Keyword Requirement
            return educationalTokens.any { lowerPkg.contains(it) || lowerLabel.contains(it) }
        }
    }

    fun handlePackageEvent(pkg: String) {
        if (pkg == deContext.packageName || pkg == "com.android.vending") return

        val isDeviceOwner = dpm?.isDeviceOwnerApp(deContext.packageName) == true

        // 1. Instant Default-Deny Quarantine (<50ms)
        if (isDeviceOwner) {
            runCatching { dpm.setPackagesSuspended(admin, arrayOf(pkg), true) }
        }

        // 2. 5-Gate Anti-Cheat Educational Verification
        val isEducational = evaluateStrictEducationalCriteria(pkg)

        // 3. Enforcement Action
        if (isEducational) {
            if (isDeviceOwner) {
                runCatching { dpm.setPackagesSuspended(admin, arrayOf(pkg), false) }
            }
            val prefs = deContext.getSharedPreferences("dynamic_study_whitelist_de", Context.MODE_PRIVATE)
            val existing = prefs.getStringSet("study_pkgs", emptySet())?.toMutableSet() ?: mutableSetOf()
            existing.add(pkg)
            prefs.edit().putStringSet("study_pkgs", existing).commit()
        } else {
            if (isDeviceOwner) {
                runCatching { dpm.setApplicationHidden(admin, pkg, true) }
            }
            val callbackIntent = Intent(deContext, PackageInstallReceiver::class.java).apply {
                action = "in.swatantra.focus.ACTION_UNINSTALL_STATUS"
            }
            val flags = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_MUTABLE
            } else {
                PendingIntent.FLAG_UPDATE_CURRENT
            }
            val pendingIntent = PendingIntent.getBroadcast(
                deContext,
                pkg.hashCode(),
                callbackIntent,
                flags
            )
            runCatching {
                pm.packageInstaller.uninstall(pkg, pendingIntent.intentSender)
            }
        }
    }

    fun evaluateStrictEducationalCriteria(pkg: String): Boolean {
        // Fast path for curated study packages
        if (pkg in curatedStudyPackages) {
            return true
        }

        // Gate B: Installer verification
        val installer = runCatching {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                pm.getInstallSourceInfo(pkg).installingPackageName
            } else {
                @Suppress("DEPRECATION")
                pm.getInstallerPackageName(pkg)
            }
        }.getOrNull()

        // Gate C: Anti-Browser & Anti-VPN intent probe
        val browserIntent = Intent(Intent.ACTION_VIEW, Uri.parse("http://www.example.com")).apply {
            addCategory(Intent.CATEGORY_BROWSABLE)
            `package` = pkg
        }
        val browserActivities = runCatching { pm.queryIntentActivities(browserIntent, 0) }.getOrDefault(emptyList())
        val hasBrowserActivity = browserActivities.isNotEmpty()

        val vpnIntent = Intent("android.net.VpnService").apply {
            `package` = pkg
        }
        val vpnServices = runCatching { pm.queryIntentServices(vpnIntent, 0) }.getOrDefault(emptyList())
        val hasVpnService = vpnServices.isNotEmpty()

        // Gate D & E: App Info & Category
        val appInfo = runCatching { pm.getApplicationInfo(pkg, 0) }.getOrNull() ?: return false
        val category = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) appInfo.category else ApplicationInfo.CATEGORY_UNDEFINED
        val appLabel = runCatching { pm.getApplicationLabel(appInfo).toString() }.getOrDefault("")

        return isStaticMetadataEducational(
            pkg = pkg,
            appLabel = appLabel,
            category = category,
            installer = installer,
            hasBrowserActivity = hasBrowserActivity,
            hasVpnService = hasVpnService
        )
    }
}

class PackageInstallReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        val action = intent.action ?: return
        if (action == Intent.ACTION_PACKAGE_ADDED || action == Intent.ACTION_PACKAGE_REPLACED) {
            val pkg = intent.data?.schemeSpecificPart ?: return
            val verifier = StrictPlayStoreAppVerifier(context)
            verifier.handlePackageEvent(pkg)
        }
    }
}

        }

class MonotonicDualClockLedger(context: Context) {
    private val deContext = context.createDeviceProtectedStorageContext()
    private val db: SQLiteDatabase = deContext.openOrCreateDatabase(
        "swatantra_ledger.db",
        Context.MODE_PRIVATE or Context.MODE_ENABLE_WRITE_AHEAD_LOGGING,
        null
    ).apply {
        execSQL("PRAGMA synchronous = FULL;")
        execSQL("""
            CREATE TABLE IF NOT EXISTS ledger_state (
                slot INTEGER PRIMARY KEY,
                seq_no INTEGER NOT NULL,
                boot_count INTEGER NOT NULL,
                boot_id TEXT NOT NULL,
                last_elapsed_ms INTEGER NOT NULL,
                high_water_wall_ms INTEGER NOT NULL,
                daily_study_ms INTEGER NOT NULL,
                daily_call_ms INTEGER NOT NULL,
                monotonic_since_reset_ms INTEGER NOT NULL,
                day_epoch INTEGER NOT NULL,
                hmac_hex TEXT NOT NULL
            )
        """.trimIndent())
    }

    @Synchronized
    fun recordHeartbeatTick(
        isStudying: Boolean,
        isOnNonWhitelistedCall: Boolean,
        verifiedTlsOrGnssTimeMs: Long?
    ): Pair<Long, Long> {
        val nowElapsed = SystemClock.elapsedRealtime()
        val rawWall = System.currentTimeMillis()
        val bootCount = Settings.Global.getInt(deContext.contentResolver, Settings.Global.BOOT_COUNT, 0)
        val bootId = runCatching { File("/proc/sys/kernel/random/boot_id").readText().trim() }.getOrDefault("unknown")

        val prev = loadLatestVerifiedSlot()
        val isSameBoot = prev != null &&
            prev.bootCount == bootCount &&
            prev.bootId == bootId &&
            nowElapsed >= prev.lastElapsedMs

        val deltaMs = if (isSameBoot) (nowElapsed - prev!!.lastElapsedMs).coerceAtLeast(0L) else nowElapsed

        val effectiveWallMs = when {
            prev == null -> verifiedTlsOrGnssTimeMs ?: rawWall
            verifiedTlsOrGnssTimeMs != null -> maxOf(prev.highWaterWallMs + deltaMs, verifiedTlsOrGnssTimeMs)
            else -> prev.highWaterWallMs + deltaMs
        }

        val candidateDayEpoch = effectiveWallMs / 86_400_000L
        val monotonicSinceReset = (prev?.monotonicSinceResetMs ?: 0L) + deltaMs

        val shouldRolloverDay = prev != null &&
            candidateDayEpoch > prev.dayEpoch &&
            (verifiedTlsOrGnssTimeMs != null || monotonicSinceReset >= 86_400_000L)

        val persistedDayEpoch = if (prev == null || shouldRolloverDay) candidateDayEpoch else prev.dayEpoch

        val nextStudyMs = (if (shouldRolloverDay) 0L else (prev?.dailyStudyMs ?: 0L)) + if (isStudying) deltaMs else 0L
        val nextCallMs = (if (shouldRolloverDay) 0L else (prev?.dailyCallMs ?: 0L)) + if (isOnNonWhitelistedCall) deltaMs else 0L
        val nextSeq = (prev?.seqNo ?: 0L) + 1L
        val targetSlot = (nextSeq % 2L).toInt()
        val nextMonotonicSinceReset = if (shouldRolloverDay) 0L else monotonicSinceReset

        val payload = "$nextSeq|$bootCount|$bootId|$nowElapsed|$effectiveWallMs|$nextStudyMs|$nextCallMs|$nextMonotonicSinceReset|$persistedDayEpoch"
        val hmac = computeHmacHex(payload)

        db.beginTransactionNonExclusive()
        try {
            db.execSQL(
                """INSERT OR REPLACE INTO ledger_state
                   (slot, seq_no, boot_count, boot_id, last_elapsed_ms, high_water_wall_ms, daily_study_ms, daily_call_ms, monotonic_since_reset_ms, day_epoch, hmac_hex)
                   VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)""",
                arrayOf<Any>(
                    targetSlot, nextSeq, bootCount, bootId, nowElapsed, effectiveWallMs,
                    nextStudyMs, nextCallMs, nextMonotonicSinceReset, persistedDayEpoch, hmac
                )
            )
            db.setTransactionSuccessful()
        } finally {
            db.endTransaction()
        }

        deContext.getSharedPreferences("call_engine_de", Context.MODE_PRIVATE)
            .edit()
            .putBoolean("daily_call_quota_exhausted", nextCallMs >= 3_600_000L)
            .commit()

        return Pair(nextStudyMs, nextCallMs)
    }

    private data class LedgerRow(
        val seqNo: Long, val bootCount: Int, val bootId: String,
        val lastElapsedMs: Long, val highWaterWallMs: Long,
        val dailyStudyMs: Long, val dailyCallMs: Long,
        val monotonicSinceResetMs: Long, val dayEpoch: Long, val hmacHex: String
    )

    private fun loadLatestVerifiedSlot(): LedgerRow? {
        db.rawQuery(
            "SELECT seq_no, boot_count, boot_id, last_elapsed_ms, high_water_wall_ms, daily_study_ms, daily_call_ms, monotonic_since_reset_ms, day_epoch, hmac_hex FROM ledger_state ORDER BY seq_no DESC",
            null
        ).use { c ->
            while (c.moveToNext()) {
                val row = LedgerRow(
                    c.getLong(0), c.getInt(1), c.getString(2), c.getLong(3),
                    c.getLong(4), c.getLong(5), c.getLong(6), c.getLong(7),
                    c.getLong(8), c.getString(9)
                )
                val payload = "${row.seqNo}|${row.bootCount}|${row.bootId}|${row.lastElapsedMs}|${row.highWaterWallMs}|${row.dailyStudyMs}|${row.dailyCallMs}|${row.monotonicSinceResetMs}|${row.dayEpoch}"
                if (computeHmacHex(payload) == row.hmacHex) return row
            }
        }
        return null
    }

    private fun computeHmacHex(payload: String): String {
        val mac = Mac.getInstance("HmacSHA256")
        mac.init(getOrCreateDirectBootHmacKey())
        return mac.doFinal(payload.toByteArray(Charsets.UTF_8)).joinToString("") { "%02x".format(it) }
    }

    private fun getOrCreateDirectBootHmacKey(): SecretKey {
        val ks = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }
        val alias = "swatantra_ledger_hmac_v7"
        (ks.getEntry(alias, null) as? KeyStore.SecretKeyEntry)?.secretKey?.let { return it }

        val keyGen = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_HMAC_SHA256, "AndroidKeyStore")
        val spec = KeyGenParameterSpec.Builder(alias, KeyProperties.PURPOSE_SIGN or KeyProperties.PURPOSE_VERIFY)
            .setUnlockedDeviceRequired(false)
            .build()
        keyGen.init(spec)
        return keyGen.generateKey()
    }
}

        // â”€â”€ Forward progression: effectiveWall must equal rawWall when ahead â”€â”€
        val futureRawWallMs  = 9999L
        val effectiveWallMs2 = maxOf(highWaterWallMs + deltaMs, futureRawWallMs)
        assertEquals(
            "When rawWall is ahead, effectiveWallMs must follow rawWall",
            9999L,
            effectiveWallMs2
        )

        // â”€â”€ 24-hour cooling-off: normal elapsed progression without reboot â”€â”€â”€â”€
        val timerStartElapsed = 1_000_000L
        val timerStartWall    = 1_700_000_000_000L
        val justBefore24h     = timerStartElapsed + 86_399_999L
        val exactlyAt24h      = timerStartElapsed + 86_400_000L

        val elapsedBefore24h = computeCoolingOffElapsedMs(timerStartElapsed, timerStartWall, justBefore24h, timerStartWall + 86_399_999L)
        assertFalse("1 ms before 24h must NOT allow PIN-free deprovision", elapsedBefore24h >= 86_400_000L)

        val elapsedAt24h = computeCoolingOffElapsedMs(timerStartElapsed, timerStartWall, exactlyAt24h, timerStartWall + 86_400_000L)
        assertTrue("At exactly 24h must allow PIN-free deprovision", elapsedAt24h >= 86_400_000L)

        // â”€â”€ 24-hour cooling-off: REBOOT SURVIVAL FALLBACK SCENARIOS â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
        // Scenario A: Device reboots after 25 hours.
        // nowElapsed reset to 10_000L (which is < requestedElapsed 1_000_000L).
        // Fallback to wall-clock: nowWall (timerStartWall + 90_000_000L) - timerStartWall = 90_000_000L >= 86_400_000L
        val postRebootElapsedA = 10_000L
        val postRebootWallA    = timerStartWall + 90_000_000L
        val elapsedAfterRebootA = computeCoolingOffElapsedMs(timerStartElapsed, timerStartWall, postRebootElapsedA, postRebootWallA)
        assertTrue("Post-reboot wall-clock fallback must permit deprovision after 24h", elapsedAfterRebootA >= 86_400_000L)

        // Scenario B: Device reboots after only 2 hours.
        // nowElapsed reset to 5_000L (< 1_000_000L).
        // Fallback to wall-clock: nowWall (timerStartWall + 7_200_000L) - timerStartWall = 7_200_000L < 86_400_000L
        val postRebootElapsedB = 5_000L
        val postRebootWallB    = timerStartWall + 7_200_000L
        val elapsedAfterRebootB = computeCoolingOffElapsedMs(timerStartElapsed, timerStartWall, postRebootElapsedB, postRebootWallB)
        assertFalse("Post-reboot wall-clock fallback must DENY deprovision before 24h", elapsedAfterRebootB >= 86_400_000L)
        assertTrue("Post-reboot timer must still register as running", elapsedAfterRebootB >= 0L)

        // Scenario C: Device reboots and RTC is set backwards (e.g. battery pull to 1970).
        // nowWall < timerStartWall -> elapsedSinceRequestMs = -1L (not running / invalid)
        val postRebootElapsedC = 2_000L
        val postRebootWallC    = timerStartWall - 50_000_000L
        val elapsedAfterRebootC = computeCoolingOffElapsedMs(timerStartElapsed, timerStartWall, postRebootElapsedC, postRebootWallC)
        assertEquals("Time-travel backwards during reboot must return -1L", -1L, elapsedAfterRebootC)

        // Scenario D: Timer was never requested (uninitialized prefs: -1L, -1L)
        val unstartedElapsed = computeCoolingOffElapsedMs(-1L, -1L, 500_000L, timerStartWall)
        assertEquals("Unstarted timer must return -1L", -1L, unstartedElapsed)
    }
